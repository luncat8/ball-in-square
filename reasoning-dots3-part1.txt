function substep(dt) {
  // integrate
  square.x += square.vx * dt;
  square.y += square.vy * dt;
  square.angle += square.omega * dt;
  ball.x += ball.vx * dt;
  ball.y += ball.vy * dt;

  const c = Math.cos(square.angle), s = Math.sin(square.angle);
  const walls = [{ox:1,oy:0},{ox:-1,oy:0},{ox:0,oy:-1},{ox:0,oy:1}];
  for (let k = 0; k < 4; k++) {
    for (const w of walls) {
      const dx = ball.x - square.x, dy = ball.y - square.y;
      const u =  c*dx + s*dy;
      const v = -s*dx + c*dy;
      let pen, cu, cv;
      if (w.ox === 1)      { pen = u - ballLimit; cu = halfl; cv = clamp(v, -halfl, halfl); }
      else if (w.ox === -1) { pen = -ballLimit - u; cu = -halfl; cv = clamp(v, -halfl, halfl); }
      else if (w.oy === -1) { pen = -ballLimit - v; cu = clamp(u, -halfl, halfl); cv = -halfl; }
      else                  { pen = v - ballLimit; cu = clamp(u, -halfl, halfl); cv = halfl; }
      if (pen <= 0) continue;

      const nx = c*w.ox - s*w.oy;
      const ny = s*w.ox + c*w.oy;
      const rx = c*cu - s*cv;
      const ry = s*cu + c*cv;

const canvas = document.getElementById('game');
const ctx = canvas.getContext('2d');
const FIELD = 1024;
const L = 200, l = 100, R = 5;
const m_s = 1, m_b = 1;
const I = (1 / 6) * (L * L + l * l) * m_s; // moment of inertia of square frame
const halfL = L / 2, halfl = l / 2, ballLimit = halfl - R;

const square = { x: 512, y: 512, vx: 0, vy: 0, angle: 0, omega: 0 };
const ball = { x: 512, y: 512, vx: 140, vy: 90 };

let allowRotation = false;

const clamp = (a, lo, hi) => a < lo ? lo : (a > hi ? hi : a);
const wrap = (v) => { const m = v % FIELD; return m < 0 ? m + FIELD : m; };

function substep(dt) {
	if (!allowRotation) square.omega = 0;
	square.x += square.vx * dt;
	square.y += square.vy * dt;
	if (allowRotation) square.angle += square.omega * dt;
	ball.x += ball.vx * dt;
	ball.y += ball.vy * dt;

	const c = Math.cos(square.angle), s = Math.sin(square.angle);
	const walls = [{ ox: 1, oy: 0 }, { ox: -1, oy: 0 }, { ox: 0, oy: -1 }, { ox: 0, oy: 1 }];
	for (let k = 0; k < 4; k++) {
		for (const w of walls) {
			const dx = ball.x - square.x, dy = ball.y - square.y;
			const u = c * dx + s * dy;
			const v = -s * dx + c * dy;
			let pen, cu, cv;
			if (w.ox === 1) { pen = u - ballLimit; cu = halfl; cv = clamp(v, -halfl, halfl); }
			else if (w.ox === -1) { pen = -ballLimit - u; cu = -halfl; cv = clamp(v, -halfl, halfl); }
			else if (w.oy === -1) { pen = -ballLimit - v; cu = clamp(u, -halfl, halfl); cv = -halfl; }
			else { pen = v - ballLimit; cu = clamp(u, -halfl, halfl); cv = halfl; }
			if (pen <= 0) continue;

			const nx = c * w.ox - s * w.oy;
			const ny = s * w.ox + c * w.oy;
			const rx = c * cu - s * cv;
			const ry = s * cu + c * cv;
			const svx = square.vx - square.omega * ry;
			const svy = square.vy + square.omega * rx;
			const rvx = ball.vx - svx;
			const rvy = ball.vy - svy;
			const vn = rvx * nx + rvy * ny;
			if (vn > 0) {
				const rCrossN = rx * ny - ry * nx;
				const denom = 1 / m_b + 1 / m_s + (rCrossN * rCrossN) / I;
				const j = 2 * vn / denom;
				ball.vx -= j * nx / m_b;
				ball.vy -= j * ny / m_b;
				square.vx += j * nx / m_s;
				square.vy += j * ny / m_s;
				if (allowRotation) square.omega += j * rCrossN / I;
				else square.omega = 0;
			}
			ball.x -= nx * pen;
			ball.y -= ny * pen;
		}
	}
}

function step(dt) {
	const sub = 4, h = dt / sub;
	for (let i = 0; i < sub; i++) substep(h);
	const nsx = wrap(square.x); ball.x += nsx - square.x; square.x = nsx;
	const nsy = wrap(square.y); ball.y += nsy - square.y; square.y = nsy;
}

	ctx.strokeStyle = '#191919';

		ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(FIELD, i); ctx.stroke();
	}

	const sx = square.x, sy = square.y;
	for (let dx = -FIELD; dx <= FIELD; dx += FIELD) {
		for (let dy = -FIELD; dy <= FIELD; dy += FIELD) {
			ctx.save();
			ctx.translate(sx + dx, sy + dy);
			ctx.rotate(square.angle);
			ctx.fillStyle = '#3a6ea5';
			ctx.fillRect(-halfL, -halfL, L, L);
			ctx.globalCompositeOperation = 'destination-out';
			ctx.fillStyle = '#000';
			ctx.fillRect(-halfl, -halfl, l, l);
			ctx.globalCompositeOperation = 'source-over';
			ctx.strokeStyle = '#cde';
			ctx.lineWidth = 2;
			ctx.strokeRect(-halfl, -halfl, l, l);
			ctx.restore();
		}
	}

	ctx.fillStyle = '#ffce3a';
	const bx = wrap(ball.x), by = wrap(ball.y);
	for (let dx = -FIELD; dx <= FIELD; dx += FIELD) {
		for (let dy = -FIELD; dy <= FIELD; dy += FIELD) {
			ctx.beginPath();
			ctx.arc(bx + dx, by + dy, R, 0, Math.PI * 2);
			ctx.fill();
		}
	}
}
